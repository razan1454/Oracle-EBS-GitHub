CREATE OR REPLACE PROCEDURE ANIL_PR_FACT(N NUMBER) IS

  S   NUMBER := N;

  RES  NUMBER := N;

  FACT_1   EXCEPTION;

  FACT_0   EXCEPTION;

BEGIN

IF S = 0 OR S = 1 
THE
RAISE FACT_1;

ELSIF S < 0 THEN
RAISE FACT_0;

END IF;

LOOP
S := S - 1;

RES := RES * S;

EXIT WHEN S = 2;

END LOOP;

DBMS_OUTPUT.PUT_LINE(' FACTORIAL = '||RES);

EXCEPTION

WHEN FACT_1 THEN
DBMS_OUTPUT.PUT_LINE(' FACTORIAL = 1 ');

WHEN FACT_0 THEN
DBMS_OUTPUT.PUT_LINE(' INPUT N CANNOT BE LESS THAN 0 ');

WHEN NO_DATA_FOUND THEN
DBMS_OUTPUT.PUT_LINE(' INPUT A NUMBER TO COMPUTE FACTORIAL');

WHEN OTHERS THEN
DBMS_OUTPUT.PUT_LINE(' CHECK YOUR PROGRAM !!! ');

END;

/
