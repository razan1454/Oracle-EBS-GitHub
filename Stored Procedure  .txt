 Registering a PL/SQL Stored Procedure in Oracle EBS

CREATE OR REPLACE PROCEDURE RAZAN_PR_TEST IS --- DB PROCEDURE ---
BEGIN
DBMS_OUTPUT.PUT_LINE(‘ WELCOME TO ORACLE EBS PROC..!’);
END;
/
----------------------------------- ORACLE EBS ---PROCEDURE------------
1. MANDATORY PARAMETERS
   a. ERRBUF OUT VARCHAR2
   b. RETCODE OUT NUMBER

2. FND_FILE.PUT_LINE(FND_OUTPUT, ‘ ORACLE EBS PROCEDURE !!’);
